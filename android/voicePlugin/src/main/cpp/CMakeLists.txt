cmake_minimum_required(VERSION 3.22)
project(dayflow_asr LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(WHISPER_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../../../../.tools/whisper/whisper.cpp-d1be6fde11ac6e0407606b4e42fe72d34add8037")
if(NOT EXISTS "${WHISPER_SOURCE}/CMakeLists.txt")
 message(FATAL_ERROR "Run python tools/fetch_whisper.py before building local speech")
endif()
add_subdirectory("${WHISPER_SOURCE}" "${CMAKE_BINARY_DIR}/whisper")
add_library(dayflow_asr SHARED bridge.cpp)
target_include_directories(dayflow_asr PRIVATE "${WHISPER_SOURCE}/include")
target_link_libraries(dayflow_asr PRIVATE whisper android log)
